⚡ Playwright Quick Reference

Locators
By Role
page.getByRole('button', { name: 'Submit' })
By Text
page.getByText('Welcome back!')
By Test‑ID
page.getByTestId('login-btn')
By Label
page.getByLabel('Email address')
By Placeholder
page.getByPlaceholder('Search…')
Interactions
Click
await locator.click()
Fill
await locator.fill('value')
Select Option
await locator.selectOption('value')
Press Key
await page.keyboard.press('Enter')
Hover
await locator.hover()
Assertions
Visible
await expect(locator).toBeVisible()
Have Text
await expect(locator).toHaveText('OK')
Have Value
await expect(input).toHaveValue('Jane')
URL
await expect(page).toHaveURL('/dashboard')
API & Network
Intercept & Abort
await page.route('**/*.png', r => r.abort())
Mock Response
await page.route('**/api/users', r => r.fulfill({ status:200, json:[{id:1}] }))
API Request
const res = await request.get('/api/users')\nawait expect(res).toBeOK()
